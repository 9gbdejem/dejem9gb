Private Sub Pausar(ByVal segundos As Double)
    Dim inicio As Double
    Dim decorrido As Double
    
    inicio = Timer
    
    Do
        If Timer >= inicio Then
            decorrido = Timer - inicio
        Else
            decorrido = (86400 - inicio) + Timer
        End If
        DoEvents
    Loop While decorrido < segundos
End Sub

Private Sub Form_Load()
    On Error Resume Next
    If Not PreencherPrazoPrimeiraEscalaFiltrada() Then
        If Nz(Me.txtDataPrazo, "") = "" Then Me.txtDataPrazo = Format(Date, "dd/mm/yyyy")
        If Nz(Me.txtHoraPrazo, "") = "" Then Me.txtHoraPrazo = Format(Time, "hh:nn")
    End If
    On Error GoTo 0
End Sub

Private Function PreencherPrazoPrimeiraEscalaFiltrada() As Boolean
    Dim frm As Form
    Dim rs As DAO.Recordset
    Dim vPrazo As Variant
    Dim dtPrazo As Date
    
    On Error GoTo Erro
    
    Set frm = ObterSubformLinhaEscalas()
    If frm Is Nothing Then Exit Function
    
    Set rs = frm.RecordsetClone
    If rs.BOF And rs.EOF Then GoTo Sair
    
    rs.MoveFirst
    Do While Not rs.EOF
        vPrazo = Nz(rs!Prazo_Inscricao, "")
        
        If Trim(CStr(vPrazo)) <> "" Then
            If IsDate(vPrazo) Then
                dtPrazo = CDate(vPrazo)
                Me.txtDataPrazo = Format(dtPrazo, "dd/mm/yyyy")
                Me.txtHoraPrazo = Format(dtPrazo, "hh:nn")
                PreencherPrazoPrimeiraEscalaFiltrada = True
                Exit Do
            End If
        End If
        
        rs.MoveNext
    Loop
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Set frm = Nothing
    On Error GoTo 0
    Exit Function
    
Erro:
    Debug.Print "Erro frmCadastrar.PreencherPrazoPrimeiraEscalaFiltrada: " & Err.Number & " - " & Err.Description
    Resume Sair
End Function

Private Sub txtDataPrazo_AfterUpdate()
    Dim strValor As String
    strValor = Trim(Nz(Me.txtDataPrazo, ""))
    
    If strValor <> "" Then
        If IsDateBrasileiro(strValor) Then
            Me.txtDataPrazo = Format(ConverterStringParaData(strValor), "dd/mm/yyyy")
        Else
            MsgBox "Data inválida. Use o formato dd/mm/aaaa (exemplo: 31/12/2026)", vbExclamation
            Me.txtDataPrazo = ""
            Me.txtDataPrazo.SetFocus
        End If
    End If
End Sub

Private Sub txtHoraPrazo_AfterUpdate()
    Dim strValor As String
    strValor = Trim(Nz(Me.txtHoraPrazo, ""))
    
    If strValor <> "" Then
        If Not NormalizarHoraPrazo() Then Me.txtHoraPrazo.SetFocus
    End If
End Sub

Private Sub txtHoraPrazo_GotFocus()
    Dim strValor As String
    
    strValor = Replace(Trim(Nz(Me.txtHoraPrazo, "")), ":", "")
    Me.txtHoraPrazo = strValor
    Me.txtHoraPrazo.SelStart = 0
    Me.txtHoraPrazo.SelLength = Len(strValor)
End Sub

Private Sub txtHoraPrazo_KeyPress(KeyAscii As Integer)
    If KeyAscii = 8 Then Exit Sub
    
    If KeyAscii < 48 Or KeyAscii > 57 Then
        KeyAscii = 0
        Beep
        Exit Sub
    End If
    
    If Len(Nz(Me.txtHoraPrazo.Text, "")) >= 4 Then
        KeyAscii = 0
        Beep
    End If
End Sub

Private Function ObterPrazoInformado(ByRef dtPrazo As Date) As Boolean
    Dim strData As String
    Dim strHora As String
    Dim partesData() As String
    
    strData = Trim(Nz(Me.txtDataPrazo, ""))
    strHora = Trim(Nz(Me.txtHoraPrazo, ""))
    
    If strData = "" Or strHora = "" Then
        MsgBox "Informe a data e o horário do prazo.", vbExclamation, "Prazo de inscrição"
        ObterPrazoInformado = False
        Exit Function
    End If
    
    If Not strData Like "##/##/####" Then
        MsgBox "Data inválida. Use o formato dd/mm/aaaa.", vbExclamation, "Prazo de inscrição"
        ObterPrazoInformado = False
        Exit Function
    End If
    
    partesData = Split(strData, "/")
    If UBound(partesData) <> 2 Then
        ObterPrazoInformado = False
        Exit Function
    End If
    
    If Not NormalizarHoraPrazo() Then
        ObterPrazoInformado = False
        Exit Function
    End If
    
    strHora = Trim(Nz(Me.txtHoraPrazo, ""))
    
    If Not IsDate(strHora) Then
        MsgBox "Horário inválido. Use o formato hh:mm.", vbExclamation, "Prazo de inscrição"
        ObterPrazoInformado = False
        Exit Function
    End If
    
    dtPrazo = DateSerial(CInt(partesData(2)), CInt(partesData(1)), CInt(partesData(0))) + TimeValue(strHora)
    ObterPrazoInformado = True
End Function

Private Function NormalizarHoraPrazo() As Boolean
    Dim strValor As String
    Dim intHora As Integer
    Dim intMinuto As Integer
    
    On Error GoTo Invalido
    
    strValor = Trim(Nz(Me.txtHoraPrazo, ""))
    strValor = Replace(strValor, ":", "")
    
    If Len(strValor) < 3 Or Len(strValor) > 4 Then GoTo Invalido
    If Not SomenteNumeros(strValor) Then GoTo Invalido
    
    If Len(strValor) = 3 Then strValor = "0" & strValor
    
    intHora = CInt(Left(strValor, 2))
    intMinuto = CInt(Right(strValor, 2))
    
    If intHora < 0 Or intHora > 23 Then GoTo Invalido
    If intMinuto < 0 Or intMinuto > 59 Then GoTo Invalido
    
    Me.txtHoraPrazo = Format(TimeSerial(intHora, intMinuto, 0), "hh:nn")
    NormalizarHoraPrazo = True
    Exit Function
    
Invalido:
    MsgBox "Horário inválido. Digite 3 ou 4 números." & vbCrLf & _
           "Exemplos: 330 vira 03:30; 1830 vira 18:30.", _
           vbExclamation, "Prazo de inscrição"
    Me.txtHoraPrazo = ""
    NormalizarHoraPrazo = False
End Function

Private Function SomenteNumeros(ByVal strValor As String) As Boolean
    Dim i As Long
    Dim strChar As String
    
    If strValor = "" Then
        SomenteNumeros = False
        Exit Function
    End If
    
    For i = 1 To Len(strValor)
        strChar = Mid(strValor, i, 1)
        If strChar < "0" Or strChar > "9" Then
            SomenteNumeros = False
            Exit Function
        End If
    Next i
    
    SomenteNumeros = True
End Function

Private Function ConverterStringParaData(ByVal strData As String) As Date
    Dim partes() As String
    Dim dia As Integer
    Dim mes As Integer
    Dim ano As Integer
    
    If strData = "" Then Exit Function
    
    partes = Split(strData, "/")
    
    If UBound(partes) = 2 Then
        dia = CInt(partes(0))
        mes = CInt(partes(1))
        ano = CInt(partes(2))
        ConverterStringParaData = DateSerial(ano, mes, dia)
    End If
End Function

Private Function IsDateBrasileiro(ByVal strData As String) As Boolean
    Dim partes() As String
    Dim dia As Integer
    Dim mes As Integer
    Dim ano As Integer
    
    If strData = "" Then
        IsDateBrasileiro = False
        Exit Function
    End If
    
    If Not strData Like "##/##/####" Then
        IsDateBrasileiro = False
        Exit Function
    End If
    
    partes = Split(strData, "/")
    
    If UBound(partes) = 2 Then
        dia = CInt(partes(0))
        mes = CInt(partes(1))
        ano = CInt(partes(2))
        
        If mes >= 1 And mes <= 12 And dia >= 1 And dia <= DiaNoMes(mes, ano) Then
            IsDateBrasileiro = True
        Else
            IsDateBrasileiro = False
        End If
    Else
        IsDateBrasileiro = False
    End If
End Function

Private Function DiaNoMes(ByVal mes As Integer, ByVal ano As Integer) As Integer
    Select Case mes
        Case 1, 3, 5, 7, 8, 10, 12
            DiaNoMes = 31
        Case 4, 6, 9, 11
            DiaNoMes = 30
        Case 2
            If (ano Mod 4 = 0 And ano Mod 100 <> 0) Or (ano Mod 400 = 0) Then
                DiaNoMes = 29
            Else
                DiaNoMes = 28
            End If
    End Select
End Function

Private Function AtualizarEscalasElegiveis(ByVal strPrazo As String, ByVal blnSubstituirPrazo As Boolean) As Long
    Dim frm As Form
    Dim rs As DAO.Recordset
    Dim db As DAO.Database
    Dim dictComp As Object
    Dim dictDados As Object
    Dim strIDCadastro As String
    Dim strComposicaoCod As String
    Dim strSQL As String
    Dim strSetPrazo As String
    Dim strWhereUpdate As String
    Dim strHoraInicial As String
    Dim lngLinha As Long
    Dim lngTotal As Long
    Dim blnElegivel As Boolean
    Dim lngStatusAdm1 As Long
    Dim lngIDEscalaVazio As Long
    Dim lngElegiveis As Long
    Dim strLinhaDebug As String
    
    On Error GoTo Erro
    
    mstrUltimoDiagnostico = ""
    
    
    On Error Resume Next
    Set frm = ObterSubformLinhaEscalas()
    If Err.Number <> 0 Then
        Debug.Print "ERRO ao executar ObterSubformLinhaEscalas: " & Err.Number & " - " & Err.Description
        mstrUltimoDiagnostico = "Erro ao localizar o subformulário: " & Err.Number & " - " & Err.Description
        Err.Clear
        On Error GoTo Erro
        GoTo Sair
    End If
    On Error GoTo Erro
    
    
    If frm Is Nothing Then
        mstrUltimoDiagnostico = "Não foi possível localizar o subformulário de linhas." & vbCrLf & _
                                "Verifique se o frmEscalas está aberto no frmApp e se o controle se chama sfrmLinhaEscala ou sfrmLinhaEscalas."
        GoTo Sair
    End If
    
    
    Set db = CurrentDb
    Set rs = frm.RecordsetClone
    
    
    If rs.BOF And rs.EOF Then
        On Error Resume Next
        rs.Close
        Set rs = Nothing
        Set rs = db.OpenRecordset(frm.RecordSource, dbOpenSnapshot)
        If Err.Number <> 0 Then
            Debug.Print "frmCadastrar.AtualizarEscalasElegiveis - Erro no fallback OpenRecordset: " & Err.Number & " - " & Err.Description
            Err.Clear
        End If
        On Error GoTo Erro
    End If
    
    Set dictComp = CarregarComposicoes()
    
    
    
    If rs.BOF And rs.EOF Then
        GoTo Sair
    End If
    
    If rs.EOF Then
        rs.MoveFirst
    End If
    
    rs.MoveLast
    lngTotal = rs.RecordCount
    
    rs.MoveFirst
    Do While Not rs.EOF
        lngLinha = lngLinha + 1
        blnElegivel = EscalaElegivel(rs)
        If Trim(Nz(rs!Status_Adm, "")) = "1" Then lngStatusAdm1 = lngStatusAdm1 + 1
        If IDEscalaVazio(rs) Then lngIDEscalaVazio = lngIDEscalaVazio + 1
        If blnElegivel Then lngElegiveis = lngElegiveis + 1
        
        strLinhaDebug = "Linha " & lngLinha & "/" & lngTotal & _
                        " | ID_Cadastro=" & Nz(rs!ID_Cadastro, "") & _
                        " | Status_Adm=" & Nz(rs!Status_Adm, "") & _
                        " | ID_Escala=" & Nz(rs!ID_Escala, "") & _
                        " | Prazo_Inscricao=" & Nz(rs!Prazo_Inscricao, "") & _
                        " | Composicao_Cod=" & Nz(rs!Composicao_Cod, "") & _
                        " | Elegivel=" & blnElegivel
        
        If lngLinha <= 10 Then
            mstrUltimoDiagnostico = mstrUltimoDiagnostico & strLinhaDebug & vbCrLf
        End If
        
        If blnElegivel Then
            strIDCadastro = Trim(Nz(rs!ID_Cadastro, ""))
            strComposicaoCod = Trim(Nz(rs!Composicao_Cod, ""))
            
            If dictComp.Exists(strComposicaoCod) Then
                Set dictDados = dictComp(strComposicaoCod)
            Else
                Set dictDados = CreateObject("Scripting.Dictionary")
            End If
            
            If blnSubstituirPrazo Or Trim(Nz(rs!Prazo_Inscricao, "")) = "" Then
                strSetPrazo = "Prazo_Inscricao = " & SQLTexto(strPrazo) & ", "
            Else
                strSetPrazo = ""
            End If
            
            If strIDCadastro <> "" Then
                strWhereUpdate = "ID_Cadastro = " & SQLTexto(strIDCadastro)
            Else
                strHoraInicial = Format(CDate(Nz(rs!Horario_Inicial, "00:00")), "hh:nn")
                strWhereUpdate = "Status_Adm = '1' " & _
                                 "AND (ID_Escala Is Null OR ID_Escala = '' OR ID_Escala = '0') " & _
                                 "AND Data_Esc = #" & Format(CDate(rs!Data_Esc), "yyyy-mm-dd") & "# " & _
                                 "AND Format(Horario_Inicial, 'hh:nn') = " & SQLTexto(strHoraInicial) & " " & _
                                 "AND Trim(Nz(Composicao_Cod,'')) = " & SQLTexto(strComposicaoCod)
            End If
            
            strSQL = "UPDATE TblEscGeral SET " & _
                     strSetPrazo & _
                     "Turno = " & SQLTexto(CalcularTurno(rs!Horario_Inicial)) & ", " & _
                     "Convenio = " & SQLTexto(DictVal(dictDados, "Convenio", "")) & ", " & _
                     "Tipo_Escala = " & SQLTexto(DictVal(dictDados, "Tipo_Escala", "")) & ", " & _
                     "AISP = " & SQLTexto(DictVal(dictDados, "AISP", "")) & ", " & _
                     "OPM_Resp = " & SQLTexto(DictVal(dictDados, "OPM_Resp", "")) & ", " & _
                     "EPI = " & SQLTexto(DictVal(dictDados, "EPI", "")) & ", " & _
                     "OBS_Intranet = " & SQLTexto(DictVal(dictDados, "OBS_Intranet", "")) & ", " & _
                     "Alteracao = 1 " & _
                     "WHERE " & strWhereUpdate
            
            db.Execute strSQL, dbFailOnError
            AtualizarEscalasElegiveis = AtualizarEscalasElegiveis + db.RecordsAffected
        End If
        
        rs.MoveNext
    Loop
    
    mstrUltimoDiagnostico = "Total no subform filtrado: " & lngTotal & vbCrLf & _
                            "Linhas com Status_Adm = 1: " & lngStatusAdm1 & vbCrLf & _
                            "Linhas com ID_Escala vazio/nulo/0: " & lngIDEscalaVazio & vbCrLf & _
                            "Linhas elegíveis: " & lngElegiveis & vbCrLf & vbCrLf & _
                            "Amostra:" & vbCrLf & mstrUltimoDiagnostico
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Set db = Nothing
    
    If Not frm Is Nothing Then frm.Requery
    Exit Function
    
Erro:
    MsgBox "Erro ao atualizar prazos: " & Err.Number & " - " & Err.Description, vbCritical
    Debug.Print "Erro frmCadastrar.AtualizarEscalasElegiveis: " & Err.Number & " - " & Err.Description
    Resume Sair
End Function

Private Function CarregarComposicoes() As Object
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim dict As Object
    Dim item As Object
    Dim strSQL As String
    Dim strCodigo As String
    
    
    Set dict = CreateObject("Scripting.Dictionary")
    Set db = CurrentDb
    
    strSQL = "SELECT CODIGO, [Convênio], TIPO, AISP, [OPM RESP], EPI, OBS FROM TblComposicao"
    
    Set rs = db.OpenRecordset(strSQL, dbOpenSnapshot)
    
    Do While Not rs.EOF
        strCodigo = Trim(Nz(rs!CODIGO, ""))
        
        Set item = CreateObject("Scripting.Dictionary")
        item.Add "Convenio", Nz(rs![Convênio], "")
        item.Add "Tipo_Escala", Nz(rs!TIPO, "")
        item.Add "AISP", Nz(rs!AISP, "")
        item.Add "OPM_Resp", Nz(rs![OPM RESP], "")
        item.Add "EPI", Nz(rs!EPI, "")
        item.Add "OBS_Intranet", Nz(rs!OBS, "")
        
        If strCodigo <> "" Then
            If dict.Exists(strCodigo) Then
                dict.Remove strCodigo
            End If
            
            dict.Add strCodigo, item
        End If
        
        rs.MoveNext
    Loop
    
    
    rs.Close
    Set rs = Nothing
    Set db = Nothing
    
    Set CarregarComposicoes = dict
End Function

Private Sub CadastrarEscalasNaIntranet()
    Dim frm As Form
    Dim rs As DAO.Recordset
    Dim driver As ChromeDriver
    Dim por As New by
    Dim strLink As String
    Dim strURL As String
    Dim strP1 As String
    Dim strCoord As String
    Dim strSubcmd As String
    Dim lngPreenchidas As Long
    Dim lngIDsImportadas As Long
    Dim lngIDsPerdidas As Long
    
    On Error GoTo Erro
    
    strLink = LinkUsuarioLogado()
    If strLink = "" Then
        MsgBox "Link do usuário logado não encontrado. Faça login novamente.", vbExclamation, "Cadastro na Intranet"
        Exit Sub
    End If
    
    strURL = URL_CAD_ESCALA & strLink
    
    Set frm = ObterSubformLinhaEscalas()
    If frm Is Nothing Then
        MsgBox "Não foi possível localizar o subformulário de linhas do frmEscalas.", _
               vbExclamation, "Cadastro na Intranet"
        GoTo Sair
    End If
    
    Set rs = frm.RecordsetClone
    
    If rs.BOF And rs.EOF Then
        MsgBox "Nenhuma escala filtrada para cadastrar.", vbInformation, "Cadastro na Intranet"
        GoTo Sair
    End If
    
    If rs.EOF Then rs.MoveFirst
    
    If ExisteEscalaForaPrazo(rs) Then
        strP1 = BuscarRESuperior(3)
        strCoord = BuscarRESuperior(2)
        strSubcmd = BuscarRESuperior(1)
        
        If strP1 = "" Or strCoord = "" Or strSubcmd = "" Then
            MsgBox "Há escala fora do prazo, mas não foi possível localizar todos os REs na tabela/lista de superiores." & _
                   vbCrLf & vbCrLf & MontarMensagemSuperiores(strP1, strCoord, strSubcmd), _
                   vbExclamation, "Cadastro na Intranet"
            GoTo Sair
        End If
    End If
    
    Set driver = New ChromeDriver
    
    With driver
        .Start
        .Window.Maximize
        .Get strURL
        Pausar 2
        
        rs.MoveFirst
        Do While Not rs.EOF
            If EscalaElegivel(rs) Then
                Call PreencherEscalaNoSite(driver, por, rs, strP1, strCoord, strSubcmd)
                lngPreenchidas = lngPreenchidas + 1
                .Refresh
                Pausar 1
            End If
            
            rs.MoveNext
        Loop
    End With
    
    SysCmd acSysCmdSetStatus, "Cadastro de escalas concluído. Baixando as IDs... aguarde."
    
    If lngPreenchidas > 0 Then
        On Error Resume Next
        driver.Quit
        Set driver = Nothing
        DoCmd.Hourglass True
        SysCmd acSysCmdSetStatus, "Baixando IDs da Intranet... aguarde."
        On Error GoTo Erro
        
        lngIDsImportadas = ImportarIDEscalasFiltradasIntranet(False, True, lngIDsPerdidas)
        
        On Error Resume Next
        SysCmd acSysCmdClearStatus
        DoCmd.Hourglass False
        On Error GoTo Erro
        
        MsgBox "Baixa de IDs concluida." & vbCrLf & _
               "ID(s) importada(s): " & lngIDsImportadas & vbCrLf & _
               "Escala(s) nao encontrada(s) gravada(s) em 'Perdidas': " & lngIDsPerdidas, _
               vbInformation, "Baixar IDs"
        
        Call AtualizarEscalasAbertasResumoFirebase(False)
    End If
    
Sair:
    On Error Resume Next
    SysCmd acSysCmdClearStatus
    DoCmd.Hourglass False
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Set driver = Nothing
    Exit Sub
    
Erro:
    If Err.Number = vbObjectError + 514 Then
        MsgBox Err.Description, vbExclamation, "Cadastro na Intranet"
    Else
        MsgBox "Erro na automa��o da Intranet: " & Err.Number & " - " & Err.Description, vbCritical
    End If
    Resume Sair
End Sub

Private Sub PreencherEscalaNoSite(ByRef driver As ChromeDriver, ByRef por As by, ByRef rs As DAO.Recordset, _
                                  ByVal strP1 As String, ByVal strCoord As String, ByVal strSubcmd As String)

Private Sub SelecionarSuperioresForaPrazo(ByRef driver As ChromeDriver, ByRef por As by, _
                                          ByVal strP1 As String, ByVal strCoord As String, ByVal strSubcmd As String)
    Dim lngTentativas As Long
    
    On Error GoTo Erro
    
    Do While Not SuperioresPreenchidos(driver, por, strP1, strCoord, strSubcmd)
        lngTentativas = lngTentativas + 1
        If lngTentativas > 30 Then
            Err.Raise vbObjectError + 513, "SelecionarSuperioresForaPrazo", _
                      "Não foi possível manter P1, Coordenador e Subcomandante preenchidos após 30 tentativas."
        End If
        
        Call SelecionarComboPorValor(driver, por, "vCOMBOP1", strP1)
        Pausar 0.5
        
        Call SelecionarComboPorValor(driver, por, "vCOMBOCOORD", strCoord)
        Pausar 0.5
        
        Call SelecionarComboPorValor(driver, por, "vCOMBOSUCMD", strSubcmd)
        Pausar 0.5
    Loop
    
    Exit Sub
    
Erro:
    Debug.Print "Erro SelecionarSuperioresForaPrazo: " & Err.Number & " - " & Err.Description
End Sub

Private Function SuperioresPreenchidos(ByRef driver As ChromeDriver, ByRef por As by, _
                                       ByVal strP1 As String, ByVal strCoord As String, ByVal strSubcmd As String) As Boolean
    SuperioresPreenchidos = (ValorElementoPorID(driver, por, "vCOMBOP1") = strP1 And _
                             ValorElementoPorID(driver, por, "vCOMBOCOORD") = strCoord And _
                             ValorElementoPorID(driver, por, "vCOMBOSUCMD") = strSubcmd)
End Function

Private Sub SelecionarComboPorValor(ByRef driver As ChromeDriver, ByRef por As by, _
                                    ByVal strID As String, ByVal strValor As String)
    driver.FindElement(por.id(strID)).AsSelect.SelectByValue (strValor)
End Sub

Private Sub AguardarAjax(ByRef driver As ChromeDriver, ByRef por As by, ByVal lngTempoMax As Long)
    Dim i As Long
    Dim blnVisivel As Boolean
    
    For i = 1 To lngTempoMax
        On Error Resume Next
        Err.Clear
        blnVisivel = driver.FindElement(por.id("gx_ajax_notification")).IsDisplayed
        If Err.Number <> 0 Then
            Err.Clear
            On Error GoTo 0
            Exit Sub
        End If
        On Error GoTo 0
        
        If Not blnVisivel Then Exit Sub
        Pausar 1
    Next
End Sub

Private Function ExisteErroDistribuicaoVagas(ByRef driver As ChromeDriver, ByRef por As by) As Boolean
    Dim strTextoErro As String
    
    On Error Resume Next
    strTextoErro = Nz(driver.FindElement(por.id("gxErrorViewer")).Attribute("innerText"), "")
    If Trim(strTextoErro) = "" Then strTextoErro = Nz(driver.FindElement(por.id("gxErrorViewer")).Text, "")
    On Error GoTo 0
    
    ExisteErroDistribuicaoVagas = (InStr(1, strTextoErro, "foram distribu", vbTextCompare) > 0 And _
                                   InStr(1, strTextoErro, "vagas", vbTextCompare) > 0) Or _
                                  (InStr(1, strTextoErro, "unidade informada", vbTextCompare) > 0 And _
                                   InStr(1, strTextoErro, "vagas distribu", vbTextCompare) > 0)
End Function

Private Function CalcularTurno(ByVal vHorario As Variant) As String
    Dim dblHora As Double
    
    If Not IsDate(vHorario) Then
        CalcularTurno = ""
        Exit Function
    End If
    
    dblHora = CDbl(TimeValue(vHorario))
    
    If dblHora < 0.1875 Then
        CalcularTurno = "Madrugada"
    ElseIf dblHora < 0.4375 Then
        CalcularTurno = "Matutino"
    ElseIf dblHora < 0.7291 Then
        CalcularTurno = "Vespertino"
    ElseIf dblHora < 0.916 Then
        CalcularTurno = "Noturno"
    Else
        CalcularTurno = "Madrugada"
    End If
End Function

Private Function FormatarDataHoraEscala(ByVal vData As Variant, ByVal vHora As Variant, _
                                        Optional ByVal blnFinal As Boolean = False, _
                                        Optional ByVal vHoraInicial As Variant) As String
    Dim dt As Date
    
    If Not IsDate(vData) Then
        FormatarDataHoraEscala = ""
        Exit Function
    End If
    
    If Not IsDate(vHora) Then
        FormatarDataHoraEscala = ""
        Exit Function
    End If
    
    dt = DateValue(vData) + TimeValue(vHora)
    
    If blnFinal Then
        If Not IsMissing(vHoraInicial) Then
            If IsDate(vHoraInicial) Then
                If TimeValue(vHora) < TimeValue(vHoraInicial) Then
                    dt = DateAdd("d", 1, dt)
                End If
            End If
        End If
    End If
    
    FormatarDataHoraEscala = Format(dt, "dd/mm/yyyy hh:nn")
End Function

Private Function FormatarHoraParte(ByVal vHora As Variant, ByVal strFormato As String) As String
    If IsDate(vHora) Then
        FormatarHoraParte = Format(TimeValue(vHora), strFormato)
    Else
        FormatarHoraParte = ""
    End If
End Function

Private Function ForaPrazo(ByVal strPrazo As String) As Boolean
    If IsDate(strPrazo) Then
        ForaPrazo = (DateAdd("d", -2, CDate(strPrazo)) < Now())
    Else
        ForaPrazo = False
    End If
End Function

Private Function ExisteEscalaForaPrazo(ByRef rs As DAO.Recordset) As Boolean
    Dim lngPosicao As Long
    
    On Error GoTo Sair
    
    If rs.BOF And rs.EOF Then Exit Function
    
    lngPosicao = rs.AbsolutePosition
    
    If rs.EOF Then rs.MoveFirst
    
    rs.MoveFirst
    Do While Not rs.EOF
        If EscalaElegivel(rs) Then
            If ForaPrazo(Nz(rs!Prazo_Inscricao, "")) Then
                ExisteEscalaForaPrazo = True
                Exit Do
            End If
        End If
        
        rs.MoveNext
    Loop
    
    If lngPosicao >= 0 Then
        rs.MoveFirst
        rs.Move lngPosicao
    Else
        rs.MoveFirst
    End If
    
Sair:
    If Err.Number <> 0 Then
        Debug.Print "Erro ExisteEscalaForaPrazo: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
End Function

Private Function BuscarRESuperior(ByVal lngSelecionado As Long) As String
    Dim rs As DAO.Recordset
    Dim strSQL As String
    Dim vSelecionado As Variant
    Dim strRE As String
    
    On Error GoTo Sair
    
    strSQL = "SELECT Selecionado, RE FROM TblListSuperiores"
    Set rs = CurrentDb.OpenRecordset(strSQL, dbOpenSnapshot)
    
    Do While Not rs.EOF
        vSelecionado = Nz(rs!Selecionado, "")
        strRE = Trim(Nz(rs!RE, ""))
        
        If strRE <> "" Then
            If Val(CStr(vSelecionado)) = lngSelecionado Then
                BuscarRESuperior = strRE
                Exit Do
            End If
        End If
        
        rs.MoveNext
    Loop
    
Sair:
    If Err.Number <> 0 Then
        Debug.Print "Erro BuscarRESuperior: " & Err.Number & " - " & Err.Description
        Err.Clear
    End If
    
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
End Function

Private Function SQLTexto(ByVal v As Variant) As String

End Function

Private Function DictVal(ByVal dict As Object, ByVal campo As String, Optional ByVal padrao As Variant = "") As Variant
    If Not dict Is Nothing Then
        If dict.Exists(campo) Then
            DictVal = dict(campo)
            Exit Function
        End If
    End If
    
    DictVal = padrao
End Function
