Option Compare Database
Option Explicit

Private Sub Form_Load()
    DoCmd.Maximize
    Forms!frmApp!frmNavegar.SourceObject = "frmDashboard"
    Forms!frmApp!frmNavbar.Form.AtualizarBotoes "frmDashboard"
End Sub

Private Sub Form_Resize()
    Dim lngCentro As Long
    Dim lngMargem As Long
    Dim lngLarguraDisponivel As Long
    
    ' Converter 1cm para twips de forma mais precisa
    ' Usando a resolução da tela atual
    lngMargem = (1440 / 2.54) ' 1440 twips/polegada ÷ 2.54 cm/polegada = aproximadamente 567 twips/cm
    ' Ou simplesmente: lngMargem = 567
    
    If Me.InsideWidth = 0 Then Exit Sub
    If Me.InsideHeight = 0 Then Exit Sub
    
    ' Calcular largura disponível (InsideWidth - margem esquerda - margem direita)
    lngLarguraDisponivel = Me.InsideWidth - (2 * lngMargem)
    
    ' Garantir que a largura não fique negativa
    If lngLarguraDisponivel < 0 Then lngLarguraDisponivel = 0
    
    ' Ajustar frmNavbar
    Me.frmNavbar.Width = lngLarguraDisponivel
    Me.frmNavbar.Left = lngMargem ' Posicionar na margem esquerda
    
    ' Ajustar frmNavegar
    Me.frmNavegar.Width = lngLarguraDisponivel
    Me.frmNavegar.Left = lngMargem ' Posicionar na margem esquerda
    
    ' Ajustar altura do frmNavegar
    Me.frmNavegar.Height = Me.InsideHeight - Me.frmNavegar.Top
End Sub
