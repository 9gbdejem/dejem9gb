Option Compare Database
Option Explicit

' Correção pontual: militar escalado na Intranet que não foi trazido ao Access.
Public Sub CorrigirEscalado8816235()
    Const ID_ESCALA As String = "8816235"
    Const RE_MILITAR As String = "157369"
    Const NOME_MILITAR As String = "ANDRE LUIS INEZ"
    Const POSTO_GRAD As String = "SOLDADO PM"
    Const OPM_MILITAR As String = "9.GB 2.SGB"

    Dim db As DAO.Database
    Dim rsEscala As DAO.Recordset
    Dim rsMilitar As DAO.Recordset
    Dim strConferencia As String
    Dim lngSup As Long
    Dim lngCap As Long
    Dim lngTen As Long
    Dim lngSgt As Long
    Dim lngCbSd As Long

    On Error GoTo TrataErro

    Set db = CurrentDb
    Set rsEscala = db.OpenRecordset( _
        "SELECT TOP 1 * FROM TblEscGeral WHERE ID_Escala = " & TextoSqlCorrecao(ID_ESCALA), _
        dbOpenSnapshot)

    If rsEscala.EOF Then
        MsgBox "A escala " & ID_ESCALA & " não foi localizada na TblEscGeral.", vbExclamation
        GoTo Finalizar
    End If

    strConferencia = ID_ESCALA & RE_MILITAR
    Set rsMilitar = db.OpenRecordset( _
        "SELECT * FROM TblMilEsc WHERE conferencia = " & TextoSqlCorrecao(strConferencia) & _
        " OR (ID_Escala = " & TextoSqlCorrecao(ID_ESCALA) & _
        " AND Militar_RE = " & TextoSqlCorrecao(RE_MILITAR) & ")", dbOpenDynaset)

    DBEngine.Workspaces(0).BeginTrans

    If rsMilitar.EOF Then
        rsMilitar.AddNew
    Else
        rsMilitar.Edit
    End If

    DefinirCampoCorrecao rsMilitar, "ID_Cadastro", Nz(rsEscala!ID_Cadastro, "")
    DefinirCampoCorrecao rsMilitar, "ID_Escala", ID_ESCALA
    DefinirCampoCorrecao rsMilitar, "Data_Esc", rsEscala!Data_Esc
    DefinirCampoCorrecao rsMilitar, "OPM_Nome", Nz(rsEscala!OPM_Nome, "")
    DefinirCampoCorrecao rsMilitar, "Composicao_Nome", Nz(rsEscala!Composicao_Nome, "")
    DefinirCampoCorrecao rsMilitar, "Composicao_Cod", Nz(rsEscala!Composicao_Cod, "")
    DefinirCampoCorrecao rsMilitar, "Horario_Inicial", rsEscala!Horario_Inicial
    DefinirCampoCorrecao rsMilitar, "Horario_Final", rsEscala!Horario_Final
    DefinirCampoCorrecao rsMilitar, "Militar_RE", RE_MILITAR
    DefinirCampoCorrecao rsMilitar, "Militar_Nome", NOME_MILITAR
    DefinirCampoCorrecao rsMilitar, "Militar_Posto_Grad", POSTO_GRAD
    DefinirCampoCorrecao rsMilitar, "Militar_OPM", OPM_MILITAR
    DefinirCampoCorrecao rsMilitar, "Status", "Concluida"
    DefinirCampoCorrecao rsMilitar, "Firebase_Enviado", "Nao"
    DefinirCampoCorrecao rsMilitar, "conferencia", strConferencia
    rsMilitar.Update

    ContarEscaladosCorrecao db, ID_ESCALA, lngSup, lngCap, lngTen, lngSgt, lngCbSd
    db.Execute "UPDATE TblEscGeral SET " & _
               "Esc_Superior = " & lngSup & ", " & _
               "Esc_Intermed = " & lngCap & ", " & _
               "Esc_Subalterno = " & lngTen & ", " & _
               "Esc_Subten_Sgt = " & lngSgt & ", " & _
               "Esc_Cb_Sd = " & lngCbSd & ", " & _
               "Alteracao = 1 " & _
               "WHERE ID_Escala = " & TextoSqlCorrecao(ID_ESCALA), dbFailOnError

    DBEngine.Workspaces(0).CommitTrans

    If Not EnviarEscaladoCorrecaoFirebase(rsEscala, ID_ESCALA, RE_MILITAR, NOME_MILITAR, POSTO_GRAD, OPM_MILITAR) Then
        MsgBox "O militar foi salvo no Access, mas o envio do escalado ao Firebase falhou." & vbCrLf & _
               "Ele continuará marcado como Nao para nova tentativa.", vbExclamation
        GoTo Finalizar
    End If

    If Not EnviarConfirmacaoCorrecaoFirebase(ID_ESCALA, RE_MILITAR) Then
        MsgBox "O militar foi salvo e enviado como escalado, mas a confirmação no Firebase falhou.", vbExclamation
        GoTo Finalizar
    End If

    db.Execute "UPDATE TblMilEsc SET Firebase_Enviado = 'Sim' " & _
               "WHERE conferencia = " & TextoSqlCorrecao(strConferencia), dbFailOnError

    MsgBox "Correção concluída." & vbCrLf & _
           "Escala: " & ID_ESCALA & vbCrLf & _
           "Militar: " & POSTO_GRAD & " - " & RE_MILITAR & " - " & NOME_MILITAR & vbCrLf & _
           "Quantidade Esc_Cb_Sd da escala: " & lngCbSd, vbInformation

Finalizar:
    On Error Resume Next
    If Not rsMilitar Is Nothing Then rsMilitar.Close
    If Not rsEscala Is Nothing Then rsEscala.Close
    Set rsMilitar = Nothing
    Set rsEscala = Nothing
    Set db = Nothing
    Exit Sub

TrataErro:
    On Error Resume Next
    DBEngine.Workspaces(0).Rollback
    Debug.Print "Erro mdlCorrigirEscalado8816235.CorrigirEscalado8816235: " & Err.Number & " - " & Err.Description
    MsgBox "Erro ao corrigir o escalado: " & Err.Number & " - " & Err.Description, vbCritical
    Resume Finalizar
End Sub

Private Sub ContarEscaladosCorrecao(ByVal db As DAO.Database, ByVal strIDEscala As String, _
                                    ByRef lngSup As Long, ByRef lngCap As Long, _
                                    ByRef lngTen As Long, ByRef lngSgt As Long, _
                                    ByRef lngCbSd As Long)
    Dim rs As DAO.Recordset
    Dim strPosto As String

    Set rs = db.OpenRecordset( _
        "SELECT Militar_Posto_Grad FROM TblMilEsc WHERE ID_Escala = " & TextoSqlCorrecao(strIDEscala), _
        dbOpenSnapshot)

    Do While Not rs.EOF
        strPosto = UCase$(Trim$(Nz(rs!Militar_Posto_Grad, "")))

        If InStr(1, strPosto, "CEL", vbTextCompare) > 0 Or InStr(1, strPosto, "MAJ", vbTextCompare) > 0 Then
            lngSup = lngSup + 1
        ElseIf InStr(1, strPosto, "CAP", vbTextCompare) > 0 Then
            lngCap = lngCap + 1
        ElseIf InStr(1, strPosto, " TEN", vbTextCompare) > 0 Then
            lngTen = lngTen + 1
        ElseIf InStr(1, strPosto, "ST", vbTextCompare) > 0 Or _
               InStr(1, strPosto, "SUBTEN", vbTextCompare) > 0 Or _
               InStr(1, strPosto, "SGT", vbTextCompare) > 0 Or _
               InStr(1, strPosto, "SARGENTO", vbTextCompare) > 0 Then
            lngSgt = lngSgt + 1
        Else
            lngCbSd = lngCbSd + 1
        End If

        rs.MoveNext
    Loop

    rs.Close
End Sub

Private Function EnviarEscaladoCorrecaoFirebase(ByVal rsEscala As DAO.Recordset, _
                                                  ByVal strIDEscala As String, _
                                                  ByVal strRE As String, _
                                                  ByVal strNome As String, _
                                                  ByVal strPosto As String, _
                                                  ByVal strOPM As String) As Boolean
    Dim dtEscala As Date
    Dim strNo As String
    Dim strChave As String
    Dim strJson As String
    Dim dblInicio As Double
    Dim dblTermino As Double

    On Error GoTo TrataErro

    If Not IsDate(rsEscala!Data_Esc) Then Exit Function
    dtEscala = DateValue(CDate(rsEscala!Data_Esc))
    dblInicio = FracaoHoraCorrecao(rsEscala!Horario_Inicial)
    dblTermino = FracaoHoraCorrecao(rsEscala!Horario_Final)
    strChave = strIDEscala & strRE
    strNo = "escalados/" & Year(dtEscala) & "/" & Format$(Month(dtEscala), "00") & "/" & Format$(Day(dtEscala), "00")

    strJson = "{"
    strJson = strJson & Chr$(34) & EscaparJsonCorrecao(strChave) & Chr$(34) & ":{"
    strJson = strJson & Chr$(34) & "Composicao" & Chr$(34) & ":" & JsonTextoCorrecao(Nz(rsEscala!Composicao_Nome, "")) & ","
    strJson = strJson & Chr$(34) & "Documento" & Chr$(34) & ":" & JsonTextoCorrecao("") & ","
    strJson = strJson & Chr$(34) & "Estacao" & Chr$(34) & ":" & JsonTextoCorrecao(Nz(rsEscala!OPM_Nome, "")) & ","
    strJson = strJson & Chr$(34) & "HorarioInic" & Chr$(34) & ":" & JsonNumeroCorrecao(dblInicio) & ","
    strJson = strJson & Chr$(34) & "HorarioTerm" & Chr$(34) & ":" & JsonNumeroCorrecao(dblTermino) & ","
    strJson = strJson & Chr$(34) & "Id" & Chr$(34) & ":" & strIDEscala & ","
    strJson = strJson & Chr$(34) & "Militar" & Chr$(34) & ":" & JsonTextoCorrecao(strNome) & ","
    strJson = strJson & Chr$(34) & "OPM" & Chr$(34) & ":" & JsonTextoCorrecao(strOPM) & ","
    strJson = strJson & Chr$(34) & "PostoGrad" & Chr$(34) & ":" & JsonTextoCorrecao(strPosto) & ","
    strJson = strJson & Chr$(34) & "RE" & Chr$(34) & ":" & strRE & ","
    strJson = strJson & Chr$(34) & "conferencia" & Chr$(34) & ":" & strChave
    strJson = strJson & "}}"

    EnviarEscaladoCorrecaoFirebase = EnviarDadosFirebasePatch(strNo, strJson)
    Exit Function

TrataErro:
    Debug.Print "Erro mdlCorrigirEscalado8816235.EnviarEscaladoCorrecaoFirebase: " & Err.Number & " - " & Err.Description
End Function

Private Function EnviarConfirmacaoCorrecaoFirebase(ByVal strIDEscala As String, ByVal strRE As String) As Boolean
    Dim strJson As String

    strJson = "{"
    strJson = strJson & Chr$(34) & "status" & Chr$(34) & ":" & JsonTextoCorrecao("concluida") & ","
    strJson = strJson & Chr$(34) & "atualizado_em" & Chr$(34) & ":" & JsonTextoCorrecao(Format$(Now, "yyyy-mm-dd\Thh:nn:ss"))
    strJson = strJson & "}"

    EnviarConfirmacaoCorrecaoFirebase = EnviarDadosFirebasePatch( _
        "confirmacoes/" & strIDEscala & "/RE_" & strRE, strJson)
End Function

Private Function FracaoHoraCorrecao(ByVal vHora As Variant) As Double
    If IsDate(vHora) Then FracaoHoraCorrecao = CDbl(TimeValue(CDate(vHora)))
End Function

Private Function JsonNumeroCorrecao(ByVal dblValor As Double) As String
    JsonNumeroCorrecao = Replace$(Format$(dblValor, "0.##########"), ",", ".")
End Function

Private Function JsonTextoCorrecao(ByVal strValor As String) As String
    JsonTextoCorrecao = Chr$(34) & EscaparJsonCorrecao(strValor) & Chr$(34)
End Function

Private Function EscaparJsonCorrecao(ByVal strTexto As String) As String
    strTexto = Replace$(strTexto, "\", "\\")
    strTexto = Replace$(strTexto, Chr$(34), "\" & Chr$(34))
    strTexto = Replace$(strTexto, vbCrLf, "\n")
    strTexto = Replace$(strTexto, vbCr, "\n")
    strTexto = Replace$(strTexto, vbLf, "\n")
    EscaparJsonCorrecao = strTexto
End Function

Private Function TextoSqlCorrecao(ByVal strValor As String) As String
    TextoSqlCorrecao = "'" & Replace$(strValor, "'", "''") & "'"
End Function

Private Sub DefinirCampoCorrecao(ByRef rs As DAO.Recordset, ByVal strCampo As String, ByVal vValor As Variant)
    On Error Resume Next
    rs.Fields(strCampo).Value = vValor
    On Error GoTo 0
End Sub
