Private Const URL_MONTAR_ESCALA As String = "http://sistemasadmin.intranet.policiamilitar.sp.gov.br/Escala/MontarEscala.aspx?"

Public Function RemoverMilitarEscalaIntranet(ByVal strIDEscala As String, ByVal strMilitarRE As String, _
                                             ByVal strConferencia As String, ByVal strDataSol As String, _
                                             ByVal strDocumento As String, ByVal strMotivo As String, _
                                             Optional ByVal strMilitarResumo As String = "") As Boolean
    Dim driver As ChromeDriver
    Dim por As New by
    Dim strLink As String
    Dim strURL As String
    Dim resposta As VbMsgBoxResult
    
    On Error GoTo Erro
    
    strIDEscala = Trim(Nz(strIDEscala, ""))
    strMilitarRE = Trim(Nz(strMilitarRE, ""))
    
    If strIDEscala = "" Or strMilitarRE = "" Then
        MsgBox "ID_Escala ou RE do militar n„o informado.", vbExclamation, "Remover militar"
        Exit Function
    End If
    
    strLink = LinkUsuarioLogado()
    If strLink = "" Then
        MsgBox "Link do usu·rio logado n„o encontrado. RefaÁa o login.", vbExclamation, "Remover militar"
        Exit Function
    End If
    
    strURL = URL_MONTAR_ESCALA & strLink
    
    DoCmd.Hourglass True
    SysCmd acSysCmdSetStatus, "Abrindo escala na Intranet... aguarde."
    
    Set driver = New ChromeDriver
    
    With driver
        .Start
        .Window.Maximize
        .Get strURL
        Pausar 1
        
        If Left(TextoElementoPorID(driver, por, "gxErrorViewer"), 14) = "Nao autorizado" Then
            MsgBox "Usu·rio n„o autorizado no sistema online. RefaÁa o login.", vbExclamation, "Remover militar"
            GoTo Sair
        End If
        
        LimparEnviarPorName driver, por, "vESCOPRIDF", strIDEscala
        .FindElement(por.Name("IMAGE2")).Click
        Pausar 0.5
        AguardarAjax driver, por, 30
        
        .FindElement(por.Name("BUTTON1")).Click
        Pausar 0.5
        AguardarTelaMilitarEscalado driver, por, 30
        
        LimparEnviarPorID driver, por, "vDLGESCHISOBS", strDocumento
    End With
    
    SysCmd acSysCmdClearStatus
    DoCmd.Hourglass False
    
    If Trim(strMilitarResumo) = "" Then strMilitarResumo = "RE " & strMilitarRE
    
    resposta = MsgBox("Realize a exclus√£o do militar " & strMilitarResumo & " na janela aberta." & vbCrLf & vbCrLf & _
                      "Ap√≥s excluir, volte ao Access e confirme abaixo." & vbCrLf & _
                      "ATEN√á√ÉO: n√£o feche a janela do navegador." & vbCrLf & vbCrLf & _
                      "A exclus√£o foi realizada?", _
                      vbQuestion + vbYesNo, "Confirmar exclus√£o")
    
    If resposta = vbYes Then
        RegistrarExclusaoLocal strIDEscala, strMilitarRE, strConferencia, strDataSol, strDocumento, strMotivo
        RemoverMilitarLocal strIDEscala, strMilitarRE, strConferencia
        RecalcularEscalados strIDEscala
        PerguntarMontarNovamente strIDEscala
        AtualizarFrmEscalasRemocao
        Call AtualizarEscalasAbertasResumoFirebase(False)
        RemoverMilitarEscalaIntranet = True
    End If
    
Sair:
    On Error Resume Next
    If Not driver Is Nothing Then driver.Quit
    SysCmd acSysCmdClearStatus
    DoCmd.Hourglass False
    Set driver = Nothing
    Exit Function
    
Erro:
    Debug.Print "Erro mdlRemoverMilitarEscala.RemoverMilitarEscalaIntranet: " & Err.Number & " - " & Err.Description
    MsgBox "Erro ao remover militar: " & Err.Number & " - " & Err.Description, vbCritical, "Remover militar"
    Resume Sair
End Function

Private Sub RegistrarExclusaoLocal(ByVal strIDEscala As String, ByVal strMilitarRE As String, _
                                   ByVal strConferencia As String, ByVal strDataSol As String, _
                                   ByVal strDocumento As String, ByVal strMotivo As String)
    Dim rsMil As DAO.Recordset
    Dim rsExc As DAO.Recordset
    Dim strSQL As String
    
    On Error GoTo Erro
    
    strSQL = "SELECT TOP 1 * FROM TblMilEsc WHERE " & WhereMilitar(strIDEscala, strMilitarRE, strConferencia)
    Set rsMil = CurrentDb.OpenRecordset(strSQL, dbOpenSnapshot)
    
    If rsMil.EOF Then
        Err.Raise vbObjectError + 930, "RegistrarExclusaoLocal", _
                  "Militar n„o encontrado na TblMilEsc para registrar exclus„o."
    End If
    
    Set rsExc = CurrentDb.OpenRecordset("TblExclusoes", dbOpenDynaset)
    
    rsExc.AddNew
    SetCampoSeExiste rsExc, "Data_Solicitacao", DataBr(strDataSol)
    SetCampoSeExiste rsExc, "Data_Exclusao", Date
    SetCampoSeExiste rsExc, "Data_Esc", Nz(rsMil!Data_Esc, Null)
    SetCampoSeExiste rsExc, "OPM_Nome", Nz(rsMil!OPM_Nome, "")
    SetCampoSeExiste rsExc, "Composicao_Nome", Nz(rsMil!Composicao_Nome, "")
    SetCampoSeExiste rsExc, "Horario_Inicial", Nz(rsMil!Horario_Inicial, Null)
    SetCampoSeExiste rsExc, "Horario_Final", Nz(rsMil!Horario_Final, Null)
    SetCampoSeExiste rsExc, "ID_Escala", Nz(rsMil!ID_Escala, "")
    SetCampoSeExiste rsExc, "Militar_RE", Nz(rsMil!Militar_RE, "")
    SetCampoSeExiste rsExc, "Militar_Nome", Nz(rsMil!Militar_Nome, "")
    SetCampoSeExiste rsExc, "Militar_Posto_Grad", Nz(rsMil!Militar_Posto_Grad, "")
    SetCampoSeExiste rsExc, "Militar_OPM", Nz(rsMil!Militar_OPM, "")
    SetCampoSeExiste rsExc, "Documento", strDocumento
    SetCampoSeExiste rsExc, "Motivo", strMotivo
    rsExc.Update
    
Sair:
    On Error Resume Next
    If Not rsMil Is Nothing Then rsMil.Close
    If Not rsExc Is Nothing Then rsExc.Close
    Set rsMil = Nothing
    Set rsExc = Nothing
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlRemoverMilitarEscala.RegistrarExclusaoLocal: " & Err.Number & " - " & Err.Description
    Err.Raise Err.Number, "RegistrarExclusaoLocal", Err.Description
End Sub

Private Sub RemoverMilitarLocal(ByVal strIDEscala As String, ByVal strMilitarRE As String, ByVal strConferencia As String)
    CurrentDb.Execute "DELETE FROM TblMilEsc WHERE " & WhereMilitar(strIDEscala, strMilitarRE, strConferencia), dbFailOnError
End Sub
Private Sub PerguntarMontarNovamente(ByVal strIDEscala As String)
    Dim resposta As VbMsgBoxResult
    
    On Error GoTo Erro
    
    resposta = MsgBox("Deseja colocar esta escala para montar novamente?", _
                      vbQuestion + vbYesNo, "Remover militar")
    
    If resposta = vbYes Then
        CurrentDb.Execute "UPDATE TblEscGeral SET Status_Adm = 3, Alteracao = 1 " & _
                          "WHERE ID_Escala = " & SQLTexto(strIDEscala), dbFailOnError
    End If
    
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlRemoverMilitarEscala.PerguntarMontarNovamente: " & Err.Number & " - " & Err.Description
    Err.Raise Err.Number, "PerguntarMontarNovamente", Err.Description
End Sub

Private Sub RecalcularEscalados(ByVal strIDEscala As String)
    Dim rs As DAO.Recordset
    Dim strSQL As String
    Dim strPosto As String
    Dim lngSup As Long
    Dim lngIntermed As Long
    Dim lngSubalterno As Long
    Dim lngSubtenSgt As Long
    Dim lngCbSd As Long
    
    On Error GoTo Erro
    
    strSQL = "SELECT Militar_Posto_Grad FROM TblMilEsc WHERE ID_Escala = " & SQLTexto(strIDEscala)
    Set rs = CurrentDb.OpenRecordset(strSQL, dbOpenSnapshot)
    
    Do While Not rs.EOF
        strPosto = Nz(rs!Militar_Posto_Grad, "")
        SomarPosto strPosto, lngSup, lngIntermed, lngSubalterno, lngSubtenSgt, lngCbSd
        rs.MoveNext
    Loop
    
    CurrentDb.Execute "UPDATE TblEscGeral SET " & _
                      "Esc_Superior = " & lngSup & ", " & _
                      "Esc_Intermed = " & lngIntermed & ", " & _
                      "Esc_Subalterno = " & lngSubalterno & ", " & _
                      "Esc_Subten_Sgt = " & lngSubtenSgt & ", " & _
                      "Esc_Cb_Sd = " & lngCbSd & ", " & _
                      "Alteracao = 1 " & _
                      "WHERE ID_Escala = " & SQLTexto(strIDEscala), dbFailOnError
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlRemoverMilitarEscala.RecalcularEscalados: " & Err.Number & " - " & Err.Description
    Resume Sair
End Sub

Private Function WhereMilitar(ByVal strIDEscala As String, ByVal strMilitarRE As String, ByVal strConferencia As String) As String
    If Trim(strConferencia) <> "" Then
        WhereMilitar = "conferencia = " & SQLTexto(strConferencia)
    Else
        WhereMilitar = "ID_Escala = " & SQLTexto(strIDEscala) & _
                       " AND Militar_RE = " & SQLTexto(strMilitarRE)
    End If
End Function

Private Sub SomarPosto(ByVal strPosto As String, ByRef lngSup As Long, ByRef lngIntermed As Long, _
                       ByRef lngSubalterno As Long, ByRef lngSubtenSgt As Long, ByRef lngCbSd As Long)
    Dim strP As String
    
    strP = UCase(Trim(Nz(strPosto, "")))
    
    If InStr(strP, "CEL") > 0 Or InStr(strP, "MAJ") > 0 Then
        lngSup = lngSup + 1
    ElseIf InStr(strP, "CAP") > 0 Then
        lngIntermed = lngIntermed + 1
    ElseIf InStr(strP, " TEN") > 0 Then
        lngSubalterno = lngSubalterno + 1
    ElseIf InStr(strP, "ST") > 0 Or InStr(strP, "SUBTEN") > 0 Or InStr(strP, "SUBTENENTE") > 0 Or InStr(strP, "SGT") > 0 Or InStr(strP, "SARGENTO") > 0 Then
        lngSubtenSgt = lngSubtenSgt + 1
    Else
        lngCbSd = lngCbSd + 1
    End If
End Sub

Private Sub AtualizarFrmEscalasRemocao()
    On Error Resume Next
    If CurrentProject.AllForms("frmEscalas").IsLoaded Then Forms!frmEscalas.btnFiltrar_Click
    If CurrentProject.AllForms("frmApp").IsLoaded Then Forms!frmApp!frmNavegar.Form.btnFiltrar_Click
    On Error GoTo 0
End Sub

Private Sub SetCampoSeExiste(ByRef rs As DAO.Recordset, ByVal strCampo As String, ByVal vValor As Variant)
    On Error Resume Next
    rs.Fields(strCampo).Value = vValor
    On Error GoTo 0
End Sub

Private Sub LimparEnviarPorName(ByRef driver As ChromeDriver, ByRef por As by, _
                                ByVal strName As String, ByVal vValor As Variant)
    With driver.FindElement(por.Name(strName))
        .Click
        .Clear
        .SendKeys CStr(Nz(vValor, ""))
    End With
End Sub

Private Sub LimparEnviarPorID(ByRef driver As ChromeDriver, ByRef por As by, _
                              ByVal strID As String, ByVal vValor As Variant)
    With driver.FindElement(por.id(strID))
        .Click
        .Clear
        .SendKeys CStr(Nz(vValor, ""))
    End With
End Sub

Private Sub AguardarAjax(ByRef driver As ChromeDriver, ByRef por As by, ByVal lngTempoMax As Long)
    Dim i As Long
    
    On Error Resume Next
    
    For i = 1 To lngTempoMax
        Err.Clear
        If driver.FindElement(por.id("gx_ajax_notification")).IsDisplayed = False Then Exit Sub
        Pausar 1
    Next
    
    If Err.Number <> 0 Then Err.Clear
End Sub

Private Sub AguardarTelaMilitarEscalado(ByRef driver As ChromeDriver, ByRef por As by, ByVal lngTempoMax As Long)
    Dim i As Long
    
    For i = 1 To lngTempoMax
        If TextoElementoPorXPath(driver, por, "//*[@id='GROUP1']/legend") = "Montar Escala: Militar Escalado" Then Exit Sub
        Pausar 1
    Next
End Sub

Private Function TextoElementoPorID(ByRef driver As ChromeDriver, ByRef por As by, ByVal strID As String) As String
    On Error GoTo Erro
    TextoElementoPorID = Trim(Nz(driver.FindElement(por.id(strID)).Text, ""))
    Exit Function
Erro:
    TextoElementoPorID = ""
End Function

Private Function TextoElementoPorXPath(ByRef driver As ChromeDriver, ByRef por As by, ByVal strXPath As String) As String
    On Error GoTo Erro
    TextoElementoPorXPath = Trim(Nz(driver.FindElement(por.XPath(strXPath)).Text, ""))
    Exit Function
Erro:
    TextoElementoPorXPath = ""
End Function

Private Function DataBr(ByVal strData As String) As Date
    Dim partes() As String
    partes = Split(strData, "/")
    DataBr = DateSerial(CInt(partes(2)), CInt(partes(1)), CInt(partes(0)))
End Function

Private Sub Pausar(ByVal segundos As Double)
    Dim inicio As Double
    Dim decorrido As Double
    
    inicio = Timer
    
    Do
        If Timer >= inicio Then
            decorrido = Timer - inicio
        Else
            decorrido = (86400 - inicio) + Timer
        End If
        DoEvents
    Loop While decorrido < segundos
End Sub

Private Function SQLTexto(ByVal valor As Variant) As String
    SQLTexto = "'" & Replace(Nz(valor, ""), "'", "''") & "'"
End Function
