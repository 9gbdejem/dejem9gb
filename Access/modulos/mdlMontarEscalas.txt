Private Const URL_MONTAR_ESCALA As String = "http://sistemasadmin.intranet.policiamilitar.sp.gov.br/Escala/MontarEscala.aspx?"

Public Sub MontarEscalasFiltradasIntranet()
    Dim frmSub As Form
    Dim rs As DAO.Recordset
    Dim driver As ChromeDriver
    Dim por As New by
    Dim strLink As String
    Dim strURL As String
    Dim lngElegiveis As Long
    Dim lngMontadas As Long
    Dim lngSemInscritos As Long
    Dim resposta As VbMsgBoxResult
    
    On Error GoTo Erro
    
    Set frmSub = ObterSubformLinhaEscala()
    If frmSub Is Nothing Then
        MsgBox "Nao foi possivel localizar o sfrmLinhaEscala.", vbExclamation, "Montar escalas"
        Exit Sub
    End If
    
    Set rs = frmSub.RecordsetClone
    If rs.BOF And rs.EOF Then
        MsgBox "Nao ha escalas exibidas para montar.", vbInformation, "Montar escalas"
        GoTo Sair
    End If
    
    rs.MoveFirst
    Do While Not rs.EOF
        If EscalaElegivelMontagem(rs) Then lngElegiveis = lngElegiveis + 1
        rs.MoveNext
    Loop
    
    If lngElegiveis = 0 Then
        MsgBox "Nenhuma escala elegivel foi encontrada." & vbCrLf & _
               "Criterios: exibida no sfrmLinhaEscala, Status_Adm = 3 e ID_Escala preenchido.", _
               vbInformation, "Montar escalas"
        GoTo Sair
    End If
    
    resposta = MsgBox("Confirma montar " & lngElegiveis & " escala(s) na intranet?", _
                      vbQuestion + vbYesNo, "Montar escalas")
    If resposta = vbNo Then GoTo Sair
    
    strLink = LinkUsuarioLogado()
    If strLink = "" Then
        MsgBox "Link do usuario logado nao encontrado. Refaça o login.", vbExclamation, "Montar escalas"
        GoTo Sair
    End If
    
    strURL = URL_MONTAR_ESCALA & strLink
    
    DoCmd.Hourglass True
    SysCmd acSysCmdSetStatus, "Montando escalas na Intranet... aguarde."
    
    Set driver = New ChromeDriver
    
    With driver
        .Start
        .Window.Maximize
        .Get strURL
        Pausar 2
    End With
    
    rs.MoveFirst
    Do While Not rs.EOF
        If EscalaElegivelMontagem(rs) Then
            If MontarUmaEscala(driver, por, rs, strURL) Then
                lngMontadas = lngMontadas + 1
            Else
                lngSemInscritos = lngSemInscritos + 1
            End If
        End If
        rs.MoveNext
    Loop
    
    MsgBox "Montagem concluida." & vbCrLf & _
           "Escala(s) elegivel(is): " & lngElegiveis & vbCrLf & _
           "Escala(s) montada(s): " & lngMontadas & vbCrLf & _
           "Sem inscritos/nao montadas: " & lngSemInscritos, _
           vbInformation, "Montar escalas"
    
    AtualizarFrmEscalas
    Call AtualizarEscalasAbertasResumoFirebase(False)
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    If Not driver Is Nothing Then driver.Quit
    SysCmd acSysCmdClearStatus
    DoCmd.Hourglass False
    Set rs = Nothing
    Set driver = Nothing
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.MontarEscalasFiltradasIntranet: " & Err.Number & " - " & Err.Description
    MsgBox "Erro ao montar escalas: " & Err.Number & " - " & Err.Description, vbCritical, "Montar escalas"
    Resume Sair
End Sub

Private Function MontarUmaEscala(ByRef driver As ChromeDriver, ByRef por As by, _
                                 ByRef rsEscala As DAO.Recordset, ByVal strURL As String) As Boolean
    Dim strIDEscala As String
    Dim lngQtdEscalados As Long
    
    On Error GoTo Erro
    
    strIDEscala = Trim(Nz(rsEscala!ID_Escala, ""))
    
    driver.Get strURL
    Pausar 0.5
    
    If TextoElementoPorID(driver, por, "gxErrorViewer") = "Nao autorizado consultar: Montar Escala" Then
        Err.Raise vbObjectError + 920, "MontarUmaEscala", _
                  "Usuario nao autorizado no sistema online. Refaça o login."
    End If
    
    LimparEnviarPorName driver, por, "vESCOPRIDF", strIDEscala
    driver.FindElement(por.Name("IMAGE2")).Click
    Pausar 0.5
    AguardarAjax driver, por, 30
    
    If Trim(TextoElementoPorID(driver, por, "gxErrorViewer")) <> "" Then Exit Function
    
    driver.FindElement(por.Name("MONESCAUT1")).Click
    Pausar 0.5
    AguardarAjax driver, por, 30
    
    lngQtdEscalados = CLng(Val(TextoElementoPorID(driver, por, "span_vTOTESC")))
    
    If lngQtdEscalados = 0 Then
        RegistrarSemInscritos rsEscala
        Exit Function
    End If
    
    driver.FindElement(por.Name("BUTTON1")).Click
    Pausar 1
    AguardarTelaMilitarEscalado driver, por, 30
    
    On Error Resume Next
    driver.FindElement(por.id("vVW_MOT_ESCEFE_ESCEFERSPPTR_0001")).AsSelect.SelectByValue "S"
    On Error GoTo Erro
    
    If Not CapturarMilitaresEscaladosComTentativas(driver, por, rsEscala, lngQtdEscalados) Then
        LimparMontagemFalha rsEscala
        Err.Raise vbObjectError + 925, "MontarUmaEscala", _
                  "Nao foi possivel capturar todos os dados dos militares da escala " & strIDEscala & _
                  " apos 3 tentativa(s)." & vbCrLf & _
                  "As linhas desta escala foram removidas da TblMilEsc, o Status_Adm voltou para 3 e as quantidades Esc_ foram zeradas."
    End If
    AtualizarEscalaMontada rsEscala
    
    AguardarAjax driver, por, 30
    
    MontarUmaEscala = True
    Exit Function
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.MontarUmaEscala ID_Escala=" & strIDEscala & ": " & Err.Number & " - " & Err.Description
    Err.Raise Err.Number, "MontarUmaEscala", Err.Description
End Function

Private Function CapturarMilitaresEscaladosComTentativas(ByRef driver As ChromeDriver, ByRef por As by, _
                                                         ByRef rsEscala As DAO.Recordset, _
                                                         ByVal lngQtdEsperada As Long) As Boolean
    Dim i As Long
    Dim strIDEscala As String
    Dim strListaREBase As String
    Dim strListaREAtual As String
    Dim strErroValidacao As String
    
    On Error GoTo Erro
    
    strIDEscala = Trim(Nz(rsEscala!ID_Escala, ""))
    
    For i = 1 To 3
        CapturarMilitaresEscalados driver, por, rsEscala
        strListaREAtual = ListaREMilitaresEscala(strIDEscala)
        
        If strListaREBase = "" And ListaREConfiavel(strListaREAtual, lngQtdEsperada) Then
            strListaREBase = strListaREAtual
        ElseIf strListaREBase <> "" And ListaREConfiavel(strListaREAtual, lngQtdEsperada) And _
               strListaREAtual <> strListaREBase Then
            strErroValidacao = "A lista de RE mudou entre as tentativas. RE inicial: " & _
                               strListaREBase & " | RE atual: " & strListaREAtual
            Exit For
        End If
        
        If MilitaresEscaladosCompletos(strIDEscala, lngQtdEsperada, strErroValidacao) Then
            CapturarMilitaresEscaladosComTentativas = True
            Exit Function
        End If
        
        Pausar 0.5
    Next
    
    Debug.Print "Falha ao capturar militares da escala " & strIDEscala & ": " & strErroValidacao
    Exit Function
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.CapturarMilitaresEscaladosComTentativas: " & Err.Number & " - " & Err.Description
    Err.Raise Err.Number, "CapturarMilitaresEscaladosComTentativas", Err.Description
End Function

Private Function ListaREConfiavel(ByVal strListaRE As String, ByVal lngQtdEsperada As Long) As Boolean
    Dim arr() As String
    Dim i As Long
    
    If Trim(strListaRE) = "" Then Exit Function
    
    arr = Split(strListaRE, "|")
    If UBound(arr) - LBound(arr) + 1 <> lngQtdEsperada Then Exit Function
    
    For i = LBound(arr) To UBound(arr)
        If Trim(arr(i)) = "" Then Exit Function
    Next
    
    ListaREConfiavel = True
End Function

Private Sub CapturarMilitaresEscalados(ByRef driver As ChromeDriver, ByRef por As by, ByRef rsEscala As DAO.Recordset)
    Dim lngLinhas As Long
    Dim i As Long
    Dim strIdx As String
    Dim strRE As String
    Dim strNome As String
    Dim strPosto As String
    Dim strOPM As String
    Dim lngSup As Long
    Dim lngIntermed As Long
    Dim lngSubalterno As Long
    Dim lngSubtenSgt As Long
    Dim lngCbSd As Long
    
    On Error GoTo Erro
    
    ExcluirMilitaresEscala Trim(Nz(rsEscala!ID_Escala, ""))
    
    lngLinhas = ContarLinhasGridMilitares(driver)
    
    For i = 1 To lngLinhas
        strIdx = Format(i, "0000")
        
        strRE = TextoElementoPorID(driver, por, "span_vVW_MOT_ESCEFE_ESCEFERE_" & strIdx)
        strNome = TextoElementoPorID(driver, por, "span_vGRIDPMNOM2_" & strIdx)
        strPosto = TextoElementoPorXPath(driver, por, "//*[@id='Grid2ContainerRow_" & strIdx & "']/td[7]")
        If strPosto = "" Then strPosto = TextoElementoPorID(driver, por, "span_vGRIDPOSDES2_" & strIdx)
        strOPM = TextoElementoPorID(driver, por, "span_vGRIDPMOPMDES2_" & strIdx)
        If strOPM = "" Then
            Pausar 0.3
            strOPM = TextoElementoPorID(driver, por, "span_vGRIDPMOPMDES2_" & strIdx)
        End If
        If strOPM = "" Then strOPM = TextoElementoPorXPath(driver, por, "//*[@id='Grid2ContainerRow_" & strIdx & "']/td[9]")
        If strOPM = "" Then strOPM = TextoElementoPorXPath(driver, por, "//*[@id='Grid2ContainerRow_" & strIdx & "']/td[10]")
        
        If Trim(strRE & strNome & strPosto & strOPM) <> "" Then
            InserirMilitarEscalado rsEscala, strRE, strNome, strPosto, strOPM
            SomarPosto strPosto, lngSup, lngIntermed, lngSubalterno, lngSubtenSgt, lngCbSd
        End If
    Next
    
    RemoverDuplicadosConferencia
    AtualizarQuantidadesEscaladas rsEscala, lngSup, lngIntermed, lngSubalterno, lngSubtenSgt, lngCbSd
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.CapturarMilitaresEscalados: " & Err.Number & " - " & Err.Description
    Err.Raise Err.Number, "CapturarMilitaresEscalados", Err.Description
End Sub

Private Function ContarLinhasGridMilitares(ByRef driver As ChromeDriver) As Long
    On Error Resume Next
    ContarLinhasGridMilitares = CLng(driver.ExecuteScript("return document.querySelectorAll('#Grid2ContainerTbl tbody tr[id^=""Grid2ContainerRow_""]').length;"))
    If Err.Number <> 0 Then
        Err.Clear
        ContarLinhasGridMilitares = 0
    End If
    On Error GoTo 0
End Function

Private Sub InserirMilitarEscalado(ByRef rsEscala As DAO.Recordset, ByVal strRE As String, _
                                   ByVal strNome As String, ByVal strPosto As String, ByVal strOPM As String)
    Dim rs As DAO.Recordset
    Dim strConferencia As String
    
    On Error GoTo Erro
    
    strConferencia = Trim(Nz(rsEscala!ID_Escala, "")) & Trim(Nz(strRE, ""))
    If RegistroConferenciaExiste(strConferencia) Then Exit Sub
    
    Set rs = CurrentDb.OpenRecordset("TblMilEsc", dbOpenDynaset)
    
    rs.AddNew
    SetCampoSeExiste rs, "ID_Cadastro", Nz(rsEscala!ID_Cadastro, "")
    SetCampoSeExiste rs, "ID_Escala", Nz(rsEscala!ID_Escala, "")
    SetCampoSeExiste rs, "Data_Esc", Nz(rsEscala!Data_Esc, Null)
    SetCampoSeExiste rs, "OPM_Nome", Nz(rsEscala!OPM_Nome, "")
    SetCampoSeExiste rs, "Composicao_Nome", Nz(rsEscala!Composicao_Nome, "")
    SetCampoSeExiste rs, "Composicao_Cod", Nz(rsEscala!Composicao_Cod, "")
    SetCampoSeExiste rs, "Horario_Inicial", Nz(rsEscala!Horario_Inicial, Null)
    SetCampoSeExiste rs, "Horario_Final", Nz(rsEscala!Horario_Final, Null)
    SetCampoSeExiste rs, "Militar_RE", strRE
    SetCampoSeExiste rs, "Militar_Nome", strNome
    SetCampoSeExiste rs, "Militar_Posto_Grad", strPosto
    SetCampoSeExiste rs, "Militar_OPM", strOPM
    SetCampoSeExiste rs, "Status", Null
    SetCampoSeExiste rs, "Firebase_Enviado", "Nao"
    SetCampoSeExiste rs, "conferencia", strConferencia
    rs.Update
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.InserirMilitarEscalado: " & Err.Number & " - " & Err.Description
    Resume Sair
End Sub

Private Function RegistroConferenciaExiste(ByVal strConferencia As String) As Boolean
    Dim rs As DAO.Recordset
    
    On Error GoTo Sair
    
    If Trim(strConferencia) = "" Then Exit Function
    
    Set rs = CurrentDb.OpenRecordset( _
        "SELECT TOP 1 conferencia FROM TblMilEsc WHERE conferencia = " & SQLTexto(strConferencia), _
        dbOpenSnapshot)
    
    RegistroConferenciaExiste = Not rs.EOF
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
End Function

Private Sub RemoverDuplicadosConferencia()
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim strSQL As String
    
    On Error GoTo Erro
    
    Set db = CurrentDb
    strSQL = "SELECT conferencia FROM TblMilEsc " & _
             "WHERE conferencia Is Not Null AND conferencia <> '' " & _
             "GROUP BY conferencia HAVING Count(*) > 1"
    
    Set rs = db.OpenRecordset(strSQL, dbOpenSnapshot)
    
    Do While Not rs.EOF
        db.Execute "DELETE FROM TblMilEsc " & _
                   "WHERE conferencia = " & SQLTexto(rs!conferencia) & " " & _
                   "AND ID NOT IN (" & _
                   "SELECT Min(ID) FROM TblMilEsc WHERE conferencia = " & SQLTexto(rs!conferencia) & _
                   ")", dbFailOnError
        rs.MoveNext
    Loop
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Set db = Nothing
    Exit Sub
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.RemoverDuplicadosConferencia: " & Err.Number & " - " & Err.Description
    Resume Sair
End Sub

Private Sub SetCampoSeExiste(ByRef rs As DAO.Recordset, ByVal strCampo As String, ByVal vValor As Variant)
    On Error Resume Next
    rs.Fields(strCampo).Value = vValor
    On Error GoTo 0
End Sub

Private Sub ExcluirMilitaresEscala(ByVal strIDEscala As String)
    On Error Resume Next
    If Trim(strIDEscala) <> "" Then
        CurrentDb.Execute "DELETE FROM TblMilEsc WHERE ID_Escala = " & SQLTexto(strIDEscala), dbFailOnError
    End If
    On Error GoTo 0
End Sub

Private Function MilitaresEscaladosCompletos(ByVal strIDEscala As String, ByVal lngQtdEsperada As Long, _
                                             ByRef strErroValidacao As String) As Boolean
    Dim rs As DAO.Recordset
    Dim strSQL As String
    Dim lngQtdGravada As Long
    
    On Error GoTo Erro
    
    If Trim(strIDEscala) = "" Then
        strErroValidacao = "ID_Escala vazio."
        Exit Function
    End If
    
    lngQtdGravada = CLng(DCount("*", "TblMilEsc", "ID_Escala = " & SQLTexto(strIDEscala)))
    If lngQtdGravada <> lngQtdEsperada Then
        strErroValidacao = "A escala retornou " & lngQtdEsperada & _
                           " militar(es) no site, mas gravou " & lngQtdGravada & " na TblMilEsc."
        GoTo Sair
    End If
    
    strSQL = "SELECT TOP 1 Militar_RE, Militar_Nome, Militar_Posto_Grad, Militar_OPM " & _
             "FROM TblMilEsc WHERE ID_Escala = " & SQLTexto(strIDEscala) & " AND (" & _
             "Militar_RE Is Null OR Militar_RE = '' OR " & _
             "Militar_Nome Is Null OR Militar_Nome = '' OR " & _
             "Militar_Posto_Grad Is Null OR Militar_Posto_Grad = '' OR " & _
             "Militar_OPM Is Null OR Militar_OPM = '')"
    
    Set rs = CurrentDb.OpenRecordset(strSQL, dbOpenSnapshot)
    
    If Not rs.EOF Then
        strErroValidacao = "Militar com campo obrigatorio vazio. RE=" & Nz(rs!Militar_RE, "") & _
                           " | Nome=" & Nz(rs!Militar_Nome, "") & _
                           " | Posto=" & Nz(rs!Militar_Posto_Grad, "") & _
                           " | OPM=" & Nz(rs!Militar_OPM, "")
        GoTo Sair
    End If
    
    MilitaresEscaladosCompletos = True
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Exit Function
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.MilitaresEscaladosCompletos: " & Err.Number & " - " & Err.Description
    Err.Raise Err.Number, "MilitaresEscaladosCompletos", Err.Description
End Function

Private Function ListaREMilitaresEscala(ByVal strIDEscala As String) As String
    Dim rs As DAO.Recordset
    Dim strLista As String
    
    On Error GoTo Erro
    
    If Trim(strIDEscala) = "" Then Exit Function
    
    Set rs = CurrentDb.OpenRecordset( _
        "SELECT Militar_RE FROM TblMilEsc WHERE ID_Escala = " & SQLTexto(strIDEscala) & _
        " ORDER BY Militar_RE", dbOpenSnapshot)
    
    Do While Not rs.EOF
        If strLista <> "" Then strLista = strLista & "|"
        strLista = strLista & Trim(Nz(rs!Militar_RE, ""))
        rs.MoveNext
    Loop
    
    ListaREMilitaresEscala = strLista
    
Sair:
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    Set rs = Nothing
    Exit Function
    
Erro:
    Debug.Print "Erro mdlMontarEscalas.ListaREMilitaresEscala: " & Err.Number & " - " & Err.Description
    Resume Sair
End Function

Private Sub AtualizarQuantidadesEscaladas(ByRef rsEscala As DAO.Recordset, _
                                          ByVal lngSup As Long, ByVal lngIntermed As Long, _
                                          ByVal lngSubalterno As Long, ByVal lngSubtenSgt As Long, _
                                          ByVal lngCbSd As Long)
    Dim strSQL As String
    Dim strWhere As String
    
    strWhere = WhereEscala(rsEscala)
    If strWhere = "" Then Exit Sub
    
    strSQL = "UPDATE TblEscGeral SET " & _
             "Esc_Superior = " & lngSup & ", " & _
             "Esc_Intermed = " & lngIntermed & ", " & _
             "Esc_Subalterno = " & lngSubalterno & ", " & _
             "Esc_Subten_Sgt = " & lngSubtenSgt & ", " & _
             "Esc_Cb_Sd = " & lngCbSd & ", " & _
             "Alteracao = 1 " & _
             "WHERE " & strWhere
    
    CurrentDb.Execute strSQL, dbFailOnError
End Sub

Private Sub AtualizarEscalaMontada(ByRef rsEscala As DAO.Recordset)
    Dim strSQL As String
    Dim strWhere As String
    
    strWhere = WhereEscala(rsEscala)
    If strWhere = "" Then Exit Sub
    
    strSQL = "UPDATE TblEscGeral SET Status_Adm = '4', Alteracao = 1 " & _
             "WHERE " & strWhere
    
    CurrentDb.Execute strSQL, dbFailOnError
End Sub

Public Sub SincronizarEscalasPendentesFirebase()
    On Error GoTo Erro

    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim strSQL As String
    Dim strIDEscala As String
    Dim strRE As String
    Dim strJson As String
    Dim strChave As String
    Dim lngEnviadas As Long
    Dim lngItensGerados As Long
    Dim lngRegistrosPendentes As Long
    Dim dictAnos As Object
    Dim dictMeses As Object
    Dim dictDias As Object
    Dim dictEscalas As Object
    Dim dictIDsPorDia As Object
    Dim varAno As Variant
    Dim varMes As Variant
    Dim varDia As Variant
    Dim varID As Variant
    Dim varIDEscala As Variant

    strSQL = "SELECT ID_Escala, Data_Esc, Militar_RE, Militar_Nome, Militar_Posto_Grad, Militar_OPM, " & _
             "OPM_Nome, Composicao_Nome, Horario_Inicial, Horario_Final " & _
             "FROM TblMilEsc " & _
             "WHERE Firebase_Enviado Is Null OR Trim(Firebase_Enviado) = '' OR UCase(Trim(Firebase_Enviado)) <> 'SIM' " & _
             "ORDER BY Data_Esc, ID_Escala, Militar_RE"

    Set db = CurrentDb
    Set rs = db.OpenRecordset(strSQL, dbOpenSnapshot)
    If rs.EOF And rs.BOF Then
        Exit Sub
    End If

    Set dictAnos = CreateObject("Scripting.Dictionary")
    Set dictIDsPorDia = CreateObject("Scripting.Dictionary")

    rs.MoveFirst
    Do While Not rs.EOF
        strIDEscala = Trim(Nz(rs!ID_Escala, ""))
        strRE = Trim(Nz(rs!Militar_RE, ""))
        If strIDEscala <> "" And strRE <> "" And IsDate(Nz(rs!Data_Esc, "")) Then
            lngRegistrosPendentes = lngRegistrosPendentes + 1

            If Not dictAnos.Exists(Year(CDate(rs!Data_Esc))) Then
                Set dictMeses = CreateObject("Scripting.Dictionary")
                dictAnos.Add Year(CDate(rs!Data_Esc)), dictMeses
            End If
            Set dictMeses = dictAnos(Year(CDate(rs!Data_Esc)))

            If Not dictMeses.Exists(Format(CDate(rs!Data_Esc), "MM")) Then
                Set dictDias = CreateObject("Scripting.Dictionary")
                dictMeses.Add Format(CDate(rs!Data_Esc), "MM"), dictDias
            End If
            Set dictDias = dictMeses(Format(CDate(rs!Data_Esc), "MM"))

            If Not dictDias.Exists(Format(CDate(rs!Data_Esc), "DD")) Then
                Set dictEscalas = CreateObject("Scripting.Dictionary")
                dictDias.Add Format(CDate(rs!Data_Esc), "DD"), dictEscalas
            End If
            Set dictEscalas = dictDias(Format(CDate(rs!Data_Esc), "DD"))

            strChave = strIDEscala & strRE
            If Not dictEscalas.Exists(strChave) Then
                strJson = MontarJsonEscalaFirebase(strIDEscala, strRE)
                If Len(strJson) > 0 Then
                    dictEscalas.Add strChave, strJson
                    lngItensGerados = lngItensGerados + 1

                    Dim strDiaKey As String
                    strDiaKey = CStr(Year(CDate(rs!Data_Esc))) & "|" & Format(CDate(rs!Data_Esc), "MM") & "|" & Format(CDate(rs!Data_Esc), "DD")
                    If Not dictIDsPorDia.Exists(strDiaKey) Then
                        Set dictEscalas = CreateObject("Scripting.Dictionary")
                        dictIDsPorDia.Add strDiaKey, dictEscalas
                    End If
                    If Not dictIDsPorDia(strDiaKey).Exists(strIDEscala) Then
                        dictIDsPorDia(strDiaKey).Add strIDEscala, True
                    End If
                End If
            End If
        Else
            Debug.Print "Registro ignorado por dados incompletos: ID_Escala=" & strIDEscala & " RE=" & strRE & " Data_Esc=" & Nz(rs!Data_Esc, "")
        End If
        rs.MoveNext
    Loop

    If lngItensGerados = 0 Then
        Debug.Print "Nenhum item valido para envio. Nao alterarei Firebase_Enviado para Sim."
        Exit Sub
    End If

    For Each varAno In dictAnos.Keys
        Set dictMeses = dictAnos(varAno)
        For Each varMes In dictMeses.Keys
            Set dictDias = dictMeses(varMes)
            For Each varDia In dictDias.Keys
                Dim strNodeDia As String
                Dim strJsonExistente As String
                Dim strJsonDiaFinal As String
                Dim strKeyDia As String

                strNodeDia = "escalados/" & CStr(varAno) & "/" & CStr(varMes) & "/" & CStr(varDia)
                strJsonExistente = BuscarJsonFirebaseNode(strNodeDia)
                strJsonDiaFinal = MesclarJsonDiaFirebase(strJsonExistente, dictDias(varDia))

                If Len(strJsonDiaFinal) > 2 Then
                    If EnviarDadosFirebasePatch(strNodeDia, strJsonDiaFinal) Then
                        strKeyDia = CStr(varAno) & "|" & CStr(varMes) & "|" & CStr(varDia)
                        If dictIDsPorDia.Exists(strKeyDia) Then
                            For Each varIDEscala In dictIDsPorDia(strKeyDia).Keys
                                db.Execute "UPDATE TblMilEsc SET Firebase_Enviado = 'Sim' WHERE ID_Escala = " & SQLTexto(CStr(varIDEscala)), dbFailOnError
                            Next
                        End If
                        lngEnviadas = lngEnviadas + 1
                    Else
                        Debug.Print "PATCH FALHOU para dia=" & varDia & " node=" & strNodeDia
                    End If
                Else
                    Debug.Print "JSON do dia vazio; nao enviarei e nao marco como Sim. node=" & strNodeDia
                End If
            Next
        Next
    Next

    Exit Sub

Erro:
    Debug.Print "Erro mdlMontarEscalas.SincronizarEscalasPendentesFirebase: " & Err.Number & " - " & Err.Description
End Sub

Private Function MontarJsonMesFirebase(ByVal dictDias As Object) As String
    Dim strJson As String
    Dim varDia As Variant
    Dim blnPrimeiroDia As Boolean

    If dictDias Is Nothing Then

        Exit Function
    End If

    strJson = "{"
    blnPrimeiroDia = True

    For Each varDia In dictDias.Keys
        If blnPrimeiroDia Then
            blnPrimeiroDia = False
        Else
            strJson = strJson & ","

    MontarJsonMesFirebase = strJson
End Function

Private Function BuscarJsonFirebaseNode(ByVal strNode As String) As String

Private Function MesclarJsonDiaFirebase(ByVal strJsonExistente As String, ByVal dictEscalasDia As Object) As String
    Dim strResultado As String
    Dim strChave As String
    Dim varChave As Variant
    Dim blnPrimeiro As Boolean

    On Error GoTo Erro

    If dictEscalasDia Is Nothing Then
        MesclarJsonDiaFirebase = "{}"
        Exit Function
    End If

    strResultado = Trim$(strJsonExistente)
    If strResultado = "" Then
        strResultado = "{}"
    End If
    If strResultado = "null" Then
        strResultado = "{}"
    End If
    If Left$(strResultado, 1) <> "{" Then
        strResultado = "{" & strResultado & "}"
    End If
    If Len(strResultado) > 2 And Right$(strResultado, 1) = "}" Then
        strResultado = Left$(strResultado, Len(strResultado) - 1)
    End If

    blnPrimeiro = (Len(strResultado) <= 1)
    If Len(strResultado) <= 1 Then
        strResultado = "{"
    End If

    For Each varChave In dictEscalasDia.Keys
        strChave = CStr(varChave)
        If Len(strChave) > 0 Then
            If InStr(1, strJsonExistente, """" & EscaparJson(strChave) & """:", vbTextCompare) = 0 Then
                If Not blnPrimeiro Then
                    strResultado = strResultado & ","
                End If
                strResultado = strResultado & """" & EscaparJson(strChave) & """:" & dictEscalasDia(varChave)
                blnPrimeiro = False
            End If
        End If
    Next

    strResultado = strResultado & "}"
    MesclarJsonDiaFirebase = strResultado
    Exit Function

Erro:
    Debug.Print "Erro MesclarJsonDiaFirebase: " & Err.Number & " - " & Err.Description
    MesclarJsonDiaFirebase = "{}"
End Function

Private Function MontarJsonEscalaFirebase(ByVal strIDEscala As String, Optional ByVal strRE As String = "") As String
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim strSQL As String
    Dim strJson As String
    Dim strDataEscala As String
    Dim strComposicao As String
    Dim strEstacao As String
    Dim strHoraInicio As String
    Dim strHoraTermino As String
    Dim strNomeMilitar As String
    Dim strPostoGrad As String
    Dim strOPMMilitar As String
    Dim strConferencia As String
    Dim dblHorarioInic As Double
    Dim dblHorarioTerm As Double

    On Error GoTo Erro

    If Trim(strIDEscala) = "" Then
        Debug.Print "MontarJsonEscalaFirebase: ID_Escala vazio."
        MontarJsonEscalaFirebase = ""
        Exit Function
    End If

    If Trim(strRE) = "" Then
        strSQL = "SELECT TOP 1 * FROM TblMilEsc WHERE ID_Escala = " & SQLTexto(strIDEscala) & " ORDER BY Militar_RE"
    Else
        strSQL = "SELECT TOP 1 * FROM TblMilEsc WHERE ID_Escala = " & SQLTexto(strIDEscala) & " AND Militar_RE = " & SQLTexto(strRE) & " ORDER BY Militar_RE"
    End If

    Set db = CurrentDb
    Set rs = db.OpenRecordset(strSQL, dbOpenSnapshot)

    If rs.EOF And rs.BOF Then
        Debug.Print "MontarJsonEscalaFirebase: registro nao encontrado | ID_Escala=" & strIDEscala & " | RE=" & strRE
        MontarJsonEscalaFirebase = ""
        Exit Function
    End If

    strDataEscala = Nz(rs!Data_Esc, "")
    strComposicao = Nz(rs!Composicao_Nome, "")
    strEstacao = Nz(rs!OPM_Nome, "")
    strNomeMilitar = Nz(rs!Militar_Nome, "")
    strPostoGrad = Nz(rs!Militar_Posto_Grad, "")
    strOPMMilitar = Nz(rs!Militar_OPM, "")
    strHoraInicio = Nz(rs!Horario_Inicial, "00:00")
    strHoraTermino = Nz(rs!Horario_Final, "00:00")

    If Not IsDate(strDataEscala) Then
        Debug.Print "MontarJsonEscalaFirebase: Data_Esc invalida | ID_Escala=" & strIDEscala & " | RE=" & strRE & " | Data_Esc=" & strDataEscala
        MontarJsonEscalaFirebase = ""
        Exit Function
    End If

    If IsDate(strHoraInicio) Then dblHorarioInic = CDbl(TimeValue(CStr(strHoraInicio)))
    If IsDate(strHoraTermino) Then dblHorarioTerm = CDbl(TimeValue(CStr(strHoraTermino)))

    strConferencia = Trim(strIDEscala) & Trim(Nz(rs!Militar_RE, ""))

    strJson = "{"
    strJson = strJson & """Composicao"":""" & EscaparJson(CStr(strComposicao)) & ""","
    strJson = strJson & """Documento"":"""","
    strJson = strJson & """Estacao"":""" & EscaparJson(CStr(strEstacao)) & ""","
    strJson = strJson & """HorarioInic"":" & Replace(Format(dblHorarioInic, "0.##########"), ",", ".") & ","
    strJson = strJson & """HorarioTerm"":" & Replace(Format(dblHorarioTerm, "0.##########"), ",", ".") & ","
    strJson = strJson & """Id"":" & Trim(strIDEscala) & ","
    strJson = strJson & """Militar"":""" & EscaparJson(CStr(strNomeMilitar)) & ""","
    strJson = strJson & """OPM"":""" & EscaparJson(CStr(strOPMMilitar)) & ""","
    strJson = strJson & """PostoGrad"":""" & EscaparJson(CStr(strPostoGrad)) & ""","
    strJson = strJson & """RE"":" & Trim(Nz(rs!Militar_RE, "0")) & ","
    strJson = strJson & """conferencia"":" & strConferencia & ","
    strJson = strJson & """mês"":" & Month(CDate(strDataEscala)) & ""
    strJson = strJson & "}"

    MontarJsonEscalaFirebase = strJson
    Exit Function

Erro:
    Debug.Print "Erro mdlMontarEscalas.MontarJsonEscalaFirebase: " & Err.Number & " - " & Err.Description & " | ID_Escala=" & strIDEscala & " | RE=" & strRE
    MontarJsonEscalaFirebase = ""
End Function

Private Function EscaparJson(ByVal strTexto As String) As String
    Dim strRetorno As String

    strRetorno = strTexto
    strRetorno = Replace(strRetorno, "\", "\\")
    strRetorno = Replace(strRetorno, Chr(34), "\" & Chr(34))
    strRetorno = Replace(strRetorno, vbCrLf, "\n")
    strRetorno = Replace(strRetorno, vbCr, "\n")
    strRetorno = Replace(strRetorno, vbLf, "\n")
    EscaparJson = strRetorno
End Function

Private Function ObservacaoSemInscritos(ByVal strAtual As String) As String

    strMsg = "Nao houve mais inscritos"
    
    If Trim(Nz(strAtual, "")) = "" Then
        ObservacaoSemInscritos = strMsg
    ElseIf InStr(1, strAtual, strMsg, vbTextCompare) = 0 Then
        ObservacaoSemInscritos = strMsg & " || " & strAtual
    Else
        ObservacaoSemInscritos = strAtual
    End If

Private Sub SomarPosto(ByVal strPosto As String, ByRef lngSup As Long, ByRef lngIntermed As Long, _
                       ByRef lngSubalterno As Long, ByRef lngSubtenSgt As Long, ByRef lngCbSd As Long)
    Dim strP As String
    
    strP = UCase(Trim(Nz(strPosto, "")))
    
    If InStr(strP, "CEL") > 0 Or InStr(strP, "MAJ") > 0 Then
        lngSup = lngSup + 1
    ElseIf InStr(strP, "CAP") > 0 Then
        lngIntermed = lngIntermed + 1
    ElseIf InStr(strP, " TEN") > 0 Then
        lngSubalterno = lngSubalterno + 1
    ElseIf InStr(strP, "ST") > 0 Or InStr(strP, "SUBTEN") > 0 Or InStr(strP, "SUBTENENTE") > 0 Or InStr(strP, "SGT") > 0 Or InStr(strP, "SARGENTO") > 0 Then
        lngSubtenSgt = lngSubtenSgt + 1
    Else
        lngCbSd = lngCbSd + 1
    End If
End Sub

Private Sub Pausar(ByVal segundos As Double)
    Dim inicio As Double
    Dim decorrido As Double
    
    inicio = Timer
    
    Do
        If Timer >= inicio Then
            decorrido = Timer - inicio
        Else
            decorrido = (86400 - inicio) + Timer
        End If
        DoEvents
    Loop While decorrido < segundos
End Sub
